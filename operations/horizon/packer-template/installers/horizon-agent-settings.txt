# -- Required -----------------------------------------------------------------

# VM is managed by vCenter Server (1=yes, 0=physical/unmanaged)
VDM_VC_MANAGED_AGENT=1

# Suppress Windows Update RunOnce check (fixes 1603 error on post-sysprep systems)
# A value of 1 skips the check for RunOnce registry keys
SUPPRESS_RUNONCE_CHECK=1

# Suppress automatic reboot - Packer handles reboots
REBOOT=ReallySuppress

# -- Features (ADDLOCAL) ------------------------------------------------------
# Core     = Required base component (must always be included)
# NGVC     = Instant Clone Agent
# USB      = USB Redirection
# RTAV     = Real-Time Audio-Video
# SmartCard= Smartcard
# TSMMR    = Windows Media Multimedia Redirection (MMR)
# HelpDesk = Horizon Help Desk Tool
ADDLOCAL=Core,NGVC,USB,RTAV,SmartCard,TSMMR,HelpDesk,ClientDriveRedirection

# -- Optional -----------------------------------------------------------------

# IP protocol (IPv4 or IPv6)
